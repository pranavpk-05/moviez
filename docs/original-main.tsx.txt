 {page==='film'&&film&&<><section className="movie-hero" style={{background:`radial-gradient(ellipse at 76% 30%,${film.color},#111315 65%)`}}><button className="back" onClick={()=>navigate('search')}><ArrowLeft size={16}/> All films</button><div className="movie-hero-inner"><span className="eyebrow">INSIDE THE CREDITS / {film.language.toUpperCase()}</span><h1>{film.title}</h1><div className="movie-meta">{film.year} <span>·</span> {film.genres.join(' / ')} <span>·</span> {film.language}</div><p>{film.synopsis}</p><button className={favourites.includes(film.id)?'btn primary':'btn secondary'} onClick={()=>toggleFilm(film.id)}><Heart size={17} fill={favourites.includes(film.id)?'currentColor':'none'}/>{favourites.includes(film.id)?'In your favourite films':'Add to favourite films'}</button></div></section><section className="section"><SectionHead eyebrow="BEHIND THE SCREEN" title="Meet the crew"/><p className="section-desc">Tap an artist to explore their credited work across cinema.</p><div className="crew-grid">{film.crew.map((c,i)=>{const p=personById(c.person);return p?<button className="crew-row" onClick={()=>navigate('person',p.id)} key={i}><span className="crew-avatar" style={{background:p.color}}>{p.initials}</span><span className="crew-names"><strong>{p.name}</strong><small>{c.role}</small></span><ArrowRight size={18}/></button>:<div className="crew-row unavailable" key={i}><span className="crew-avatar"><Users size={18}/></span><span className="crew-names"><strong>{c.person.split('-').map(v=>v[0].toUpperCase()+v.slice(1)).join(' ')}</strong><small>{c.role} · Profile coming soon</small></span></div>})}</div><div className="editorial-note"><Info size={18}/> Seed filmography for prototype demonstration. Independently verify all credits before publishing this site as an authoritative database.</div></section></>}
 {page==='taste'&&<section className="section page taste-page"><span className="eyebrow">YOUR CINEMATIC FINGERPRINT</span><h1 className="page-title">Follow your <em>instincts.</em></h1><p className="page-intro">Tell us what you love. We'll connect you to the craftspeople behind those creative experiences.</p><div className="step-title"><span>01</span><div><h2>Which films stay with you?</h2><p>Choose a few favourites.  {page==='saved'&&<section className="section page"><span className="eyebrow">YOUR PERSONAL COLLECTION</span><h1 className="page-title">Artists worth <em>remembering.</em></h1><p className="page-intro">The creative voices you want to come back to.
