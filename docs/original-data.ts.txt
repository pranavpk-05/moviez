// Unresolved collaborators are intentionally listed on films but never presented as completed person profiles.
export const personById=(id:string)=>people.find(p=>p.id===id);
export const filmById=(id:string)=>films.find(f=>f.id===id);
export const creditsFor=(id:string)=>films.filter(f=>f.crew.some(c=>c.person===id));
export const filmStyles=(f:Film)=>Array.from(new Set(f.crew.flatMap(c=>personById(c.person)?.styles||[])));
export type Recommendation={person:Person;score:number;matched:string[];references:Film[];reason:string};
export function recommend(favourites:string[],chosenStyles:string[],dept:string):Recommendation[]{
 const seen=new Set(favourites); const favFilms=films.filter(f=>seen.has(f.id));
 return people.filter(p=>dept==='All departments'||p.department===dept).map(person=>{
 const references=creditsFor(person.id); const matched=chosenStyles.filter(s=>person.styles.includes(s));
 const direct=references.filter(f=>seen.has(f.id));
 const similarGenres=references.reduce((n,f)=>n+(f.genres.some(g=>favFilms.some(ff=>ff.genres.includes(g)))?1:0),0);
 const score=matched.length*24+direct.length*30+Math.min(similarGenres,3)*6+Math.min(references.length,5)*2;
 const evidence=direct.length?`You liked ${direct.slice(0,2).map(f=>f.title).join(' and ')}, which credit ${person.name}.`:references.length?`Explore their credited work on ${references.slice(0,2).map(f=>f.title).join(' and ')}.`:`Discover this ${person.department.toLowerCase()} professional.`;
 const reason=(matched.length?`Matches your interest in ${matched.slice(0,2).join(' and ').toLowerCase()}. `:'')+evidence;
 return {person,score,matched,references,reason};
 }).filter(r=>r.references.length>0).sort((a,b)=>b.score-a.score||a.person.name.localeCompare(b.person.name));
}
